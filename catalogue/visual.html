// --- STAGE 1: Gemini Vision Sleeve Inspection (Latest Gemini 3.8 Flash) --- async function extractVisualCluesWithGemini(apiKey, base64Jpg) { const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${apiKey}`; const prompt = `Inspect this album cover, vinyl sleeve, or CD insert closely. Extract all release-identifying details: 1. "artist": Artist or band name 2. "album": Official release title 3. "catno": Catalog number or barcode etched or printed on the sleeve (empty string if none) 4. "label": Record label name (e.g. Go! Discs, Big Dada, Ninja Tune, Factory, 4AD, Warp) 5. "year": Stated copyright or release year (empty string if not visible) 6. "format": Inferred format (e.g. "Vinyl, LP", "CD, Album", "12 Inch") 7. "country": Country of manufacture/distribution if printed (e.g. UK, Europe, US) Return strictly valid JSON: { "artist": "string", "album": "string", "catno": "string", "label": "string", "year": "string", "format": "string", "country": "string" }`; const payload = { contents: [{ parts: [ { text: prompt }, { inline_data: { mime_type: "image/jpeg", data: base64Jpg } } ] }], generationConfig: { response_mime_type: "application/json", temperature: 0.1 } }; let res; try { res = await fetch(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); } catch (networkErr) { throw new Error('Connection failed. Verify API key and network.'); } if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error?.message || `Gemini API HTTP ${res.status}`); } const data = await res.json(); const raw = data.candidates?.[0]?.content?.parts?.[0]?.text; if (!raw) throw new Error('No recognition data returned.'); return JSON.parse(raw); }
